cmake_minimum_required(VERSION 3.22)
project(MineCloneAndroid LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain or tools/build-android.sh")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PLATFORM Android CACHE STRING "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# Reuse the pinned sources downloaded by the Windows build when available.
set(RAYLIB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/windows/sources/raylib-5.5" CACHE PATH "Raylib 5.5 source tree")
if(EXISTS "${RAYLIB_SOURCE_DIR}/CMakeLists.txt")
    add_subdirectory("${RAYLIB_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/raylib")
else()
    include(FetchContent)
    FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e)
    FetchContent_MakeAvailable(raylib)
endif()
# Raylib 5.5's ES default vertex shader uses mediump. World-space entity
# vertices lose precision on mobile GPUs, especially when multiplied by the view
# matrix. Patch an Android-only generated copy, leaving desktop sources intact.
get_target_property(raylib_source raylib SOURCE_DIR)
set(mobile_raylib "${CMAKE_CURRENT_BINARY_DIR}/mobile-raylib")
file(MAKE_DIRECTORY "${mobile_raylib}")
file(READ "${raylib_source}/rlgl.h" mobile_rlgl)
string(FIND "${mobile_rlgl}" "const char *defaultFShaderCode" fragment_offset)
if(fragment_offset LESS 0)
    message(FATAL_ERROR "Raylib default shader layout changed; review mobile precision patch")
endif()
string(SUBSTRING "${mobile_rlgl}" 0 ${fragment_offset} vertex_part)
string(SUBSTRING "${mobile_rlgl}" ${fragment_offset} -1 fragment_part)
string(REPLACE "precision mediump float;" "precision highp float;  " vertex_part "${vertex_part}")
# Keep interpolated UV/color precision compatible with the fragment shader.
string(REPLACE "varying vec" "varying mediump vec" vertex_part "${vertex_part}")
file(WRITE "${mobile_raylib}/rlgl.h" "${vertex_part}${fragment_part}")
file(READ "${raylib_source}/rcore.c" mobile_rcore)
string(REPLACE "#include \"rlgl.h\"" "#include \"${mobile_raylib}/rlgl.h\"" mobile_rcore "${mobile_rcore}")
file(WRITE "${mobile_raylib}/rcore.c" "${mobile_rcore}")
set_source_files_properties("${raylib_source}/rcore.c" TARGET_DIRECTORY raylib PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(raylib PRIVATE "${mobile_raylib}/rcore.c")
target_include_directories(raylib PRIVATE "${raylib_source}")

file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../src/*.cpp")
add_library(mineclone SHARED ${GAME_SOURCES})
target_include_directories(mineclone PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include" "${ANDROID_NDK}/sources/android/native_app_glue")
target_compile_definitions(mineclone PRIVATE PLATFORM_ANDROID GRAPHICS_API_OPENGL_ES2)
target_compile_options(mineclone PRIVATE -O2 -Wall -Wextra)
target_link_options(mineclone PRIVATE -u ANativeActivity_onCreate -Wl,-z,max-page-size=16384)
target_link_libraries(mineclone PRIVATE raylib z log android EGL GLESv2 OpenSLES dl atomic)
